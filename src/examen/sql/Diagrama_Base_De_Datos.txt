

Table clientes {
  id int [pk, increment]
  nombre varchar(100) [not null]
  email varchar(100) [unique, not null]
  telefono varchar(20)
  estado enum('ACTIVO', 'INACTIVO') [default: 'ACTIVO', not null]
  limite_credito decimal(10, 2) [default: 0.00, not null]
  created_at datetime [default: `CURRENT_TIMESTAMP`]
}

Table direcciones {
  id int [pk, increment]
  cliente_id int [not null]
  alias varchar(50) [note: 'Ej. Casa, Oficina, Bodega']
  calle varchar(150) [not null]
  ciudad varchar(100) [not null]
  estado varchar(100) [not null]
  codigo_postal varchar(10) [not null]
  es_principal boolean [default: false]
}

Table almacenes {
  id int [pk, increment]
  nombre varchar(100) [not null]
  ubicacion varchar(200) [not null]
  capacidad_maxima int [not null, note: 'Validado por trigger before_insert_paquete_almacen']
}

Table envios {
  id int [pk, increment]
  remitente_id int [not null]
  destinatario_id int [not null]
  direccion_origen_id int [not null]
  direccion_destino_id int [not null]
  costo_total decimal(10, 2) [not null]
  fecha datetime [default: `CURRENT_TIMESTAMP`, not null]
  estado_global enum('REGISTRADO', 'EN_TRANSITO', 'ENTREGADO', 'CANCELADO') [default: 'REGISTRADO', not null]
}

Table paquetes {
  id int [pk, increment]
  envio_id int [not null]
  almacen_id int [null, note: 'Nullable al crearse el envío; se asigna en POST /api/almacenes/:id/paquetes']
  peso decimal(8, 2) [not null]
  dimensiones varchar(50) [not null, note: 'Ej. 30x20x15 cm']
  contenido varchar(255) [not null]
}

Table historial_rastreo {
  id int [pk, increment]
  envio_id int [not null]
  fecha datetime [default: `CURRENT_TIMESTAMP`, not null]
  ubicacion_actual varchar(150) [not null]
  estado_evento enum('REGISTRADO', 'EN_ALMACEN', 'EN_TRANSITO', 'ENTREGADO', 'INCIDENCIA') [not null]
  observaciones text
}

Table auditoria_envios {
  id int [pk, increment]
  envio_id int [not null]
  costo_anterior decimal(10, 2) [not null]
  costo_nuevo decimal(10, 2) [not null]
  usuario varchar(100) [not null, note: 'Capturado con CURRENT_USER() en el trigger']
  fecha_modificacion datetime [default: `CURRENT_TIMESTAMP`, not null]
}



// Un cliente tiene muchas direcciones
Ref: direcciones.cliente_id > clientes.id

// Un envío tiene un remitente y un destinatario (ambos son clientes)
Ref: envios.remitente_id > clientes.id
Ref: envios.destinatario_id > clientes.id

// Un envío tiene una dirección de origen y una de destino
Ref: envios.direccion_origen_id > direcciones.id
Ref: envios.direccion_destino_id > direcciones.id

// Un envío contiene muchos paquetes
Ref: paquetes.envio_id > envios.id

// Un almacén resguarda muchos paquetes
Ref: paquetes.almacen_id > almacenes.id

// Un envío tiene múltiples registros en su bitácora de rastreo
Ref: historial_rastreo.envio_id > envios.id

// Un envío puede tener múltiples registros de auditoría por cambios de costo
Ref: auditoria_envios.envio_id > envios.id
